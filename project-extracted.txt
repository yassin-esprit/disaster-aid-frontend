export class AlertDetailComponent implements OnInit {
  alert?: Alert;
  isNew     = false;
  editMode  = false;

  disasterTypes = ['FLOOD', 'EARTHQUAKE', 'WILDFIRE', 'LANDSLIDE', 'STORM', 'CHEMICAL'];
  severities    = ['LOW', 'MODERATE', 'HIGH', 'CRITICAL'];

  formData: any = { title: '', disasterType: 'FLOOD', severity: 'MODERATE', description: '', confidenceScore: null, issuedBy: 1 };
  editData: any = {};

  get mapGeometries() {
    const geoms: any[] = [];
    if (this.alert?.affectedArea)    geoms.push({ type: 'Polygon', wkt: this.alert.affectedArea,    color: '#ff3388' });
    if (this.alert?.predictedArea2h) geoms.push({ type: 'Polygon', wkt: this.alert.predictedArea2h, color: '#ffaa33' });
    if (this.alert?.predictedArea6h) geoms.push({ type: 'Polygon', wkt: this.alert.predictedArea6h, color: '#33aaff' });
    return geoms;
  }

  constructor(private route: ActivatedRoute, private router: Router, private alertService: AlertService) {}

  ngOnInit(): void {
    this.isNew = window.location.href.includes('/new');
    if (!this.isNew) {
      const id = Number(this.route.snapshot.paramMap.get('id'));
      if (!isNaN(id) && id > 0) {
        this.alertService.getAlertById(id).subscribe(data => {
          this.alert    = data;
          this.editData = { ...data };
        });
      }
    }
  }

  createAlert(): void  { this.alertService.createAlert(this.formData).subscribe(() => this.router.navigate(['/alerts'])); }
  enableEdit(): void   { this.editMode = true; this.editData = { ...this.alert }; }
  cancelEdit(): void   { this.editMode = false; }
  updateAlert(): void  {
    if (!this.alert) return;
    this.alertService.updateAlert(this.alert.id, this.editData).subscribe(u => { this.alert = u; this.editMode = false; });
  }
  deleteAlert(): void {
    if (!this.alert) return;
    if (confirm('Delete this alert?')) {
      this.alertService.deleteAlert(this.alert.id).subscribe({ next: () => this.router.navigate(['/alerts']) });
    }
  }
  resolve(): void {
    if (this.alert) this.alertService.resolveAlert(this.alert.id).subscribe(u => this.alert = u);
  }
  severityClass(s: string): string {
    return { CRITICAL: 'bg-error text-white', HIGH: 'bg-orange-500/20 text-orange-400', MODERATE: 'bg-amber-500/20 text-amber-400', LOW: 'bg-green-500/20 text-green-400' }[s] ?? '';
  }
}

export class AlertListComponent implements OnInit {
  alerts: Alert[]  = [];
  filtered: Alert[] = [];
  searchTerm     = '';
  severityFilter = '';
  statusFilter   = '';

  constructor(private alertService: AlertService) {}

  ngOnInit(): void {
    this.alertService.getAllAlerts().subscribe(data => {
      this.alerts   = data;
      this.filtered = data;
    });
  }

  applyFilters(): void {
    this.filtered = this.alerts.filter(a => {
      const matchSearch   = !this.searchTerm     ||
        (a.title || '').toLowerCase().includes(this.searchTerm.toLowerCase()) ||
        a.disasterType.toLowerCase().includes(this.searchTerm.toLowerCase());
      const matchSeverity = !this.severityFilter || a.severity     === this.severityFilter;
      const matchStatus   = !this.statusFilter   || a.status       === this.statusFilter;
      return matchSearch && matchSeverity && matchStatus;
    });
  }

  rowClass(severity: string): string {
    return severity === 'CRITICAL' ? 'border-l-2 border-l-error' :
      severity === 'HIGH'     ? 'border-l-2 border-l-orange-500' : '';
  }

  severityClass(severity: string): string {
    return {
      CRITICAL: 'bg-error text-white',
      HIGH:     'bg-orange-500/20 text-orange-400 border border-orange-500/30',
      MODERATE: 'bg-amber-500/20 text-amber-400 border border-amber-500/30',
      LOW:      'bg-green-500/20 text-green-400 border border-green-500/30',
    }[severity] ?? 'bg-white/10 text-white';
  }

  disasterIcon(type: string): string {
    const m: Record<string,string> = {
      FLOOD: 'water', EARTHQUAKE: 'earthquake', WILDFIRE: 'local_fire_department',
      LANDSLIDE: 'landslide', STORM: 'thunderstorm', CHEMICAL: 'science'
    };
    return m[type] ?? 'emergency';
  }
  disasterIconBg(type: string): string {
    const m: Record<string,string> = {
      FLOOD: 'bg-blue-500/20', EARTHQUAKE: 'bg-orange-500/20', WILDFIRE: 'bg-red-500/20',
      LANDSLIDE: 'bg-amber-500/20', STORM: 'bg-sky-500/20', CHEMICAL: 'bg-purple-500/20'
    };
    return m[type] ?? 'bg-white/10';
  }
  disasterIconColor(type: string): string {
    const m: Record<string,string> = {
      FLOOD: 'text-blue-400', EARTHQUAKE: 'text-orange-400', WILDFIRE: 'text-red-400',
      LANDSLIDE: 'text-amber-400', STORM: 'text-sky-400', CHEMICAL: 'text-purple-400'
    };
    return m[type] ?? 'text-white';
  }

export class EvacuationRoutesComponent implements OnInit, AfterViewInit, OnDestroy {
  @ViewChild('pickerMap') pickerMapRef!: ElementRef;

  routes: EvacuationRoute[] = [];
  filtered: EvacuationRoute[] = [];
  alerts: Alert[] = [];
  statusFilter = '';
  showForm = false;
  creating = false;
  loading = true;
  formError = '';
  pinMode: 'origin' | 'shelter' = 'origin';
  newRoute: Partial<EvacuationRoute> = {};

  private pickerMap!: L.Map;
  private pickerOriginMarker: L.Marker | null = null;
  private pickerShelterMarker: L.Marker | null = null;
  private cardMaps: Map<number, L.Map> = new Map();
  private pickerMapInit = false;

  constructor(
    private evacuationService: EvacuationService,
    private alertService: AlertService,
    private cdr: ChangeDetectorRef
  ) {}

  ngOnInit(): void {
    this.load();
    this.alertService.getAllAlerts().subscribe(a => this.alerts = a);
  }

  ngAfterViewInit(): void {
    setTimeout(() => this.initCardMaps(), 600);
  }

  ngOnDestroy(): void {
    this.cardMaps.forEach(m => m.remove());
    if (this.pickerMap) this.pickerMap.remove();
  }

  toggleForm(): void {
    this.showForm = !this.showForm;
    if (this.showForm) {
      this.newRoute = {};
      this.pinMode = 'origin';
      this.pickerMapInit = false;
      setTimeout(() => this.initPickerMap(), 150);
    } else {
      if (this.pickerMap) { this.pickerMap.remove(); this.pickerMapInit = false; }
    }
  }

  load(): void {
    this.loading = true;
    this.evacuationService.getAllRoutes().subscribe({
      next: data => {
        this.routes = data;
        this.filtered = data;
        this.loading = false;
        setTimeout(() => this.initCardMaps(), 400);
      },
      error: () => { this.loading = false; }
    });
  }

  applyFilters(): void {
    this.filtered = this.statusFilter
      ? this.routes.filter(r => r.status === this.statusFilter)
      : [...this.routes];
    setTimeout(() => this.initCardMaps(), 300);
  }

  createRoute(): void {
    this.formError = '';
    if (!this.newRoute.alertId) { this.formError = 'Please select an alert.'; return; }
    if (!this.newRoute.shelterName?.trim()) { this.formError = 'Please enter a shelter name.'; return; }
    if (!this.newRoute.fromLatitude || !this.newRoute.fromLongitude) { this.formError = 'Please pin the origin on the map.'; return; }
    if (!this.newRoute.toLatitude || !this.newRoute.toLongitude) { this.formError = 'Please pin the shelter on the map.'; return; }
    this.creating = true;
    this.evacuationService.createRoute(this.newRoute).subscribe({
      next: () => { this.creating = false; this.toggleForm(); this.load(); },
      error: (err) => { this.creating = false; this.formError = err?.error?.message || 'Failed to create route.'; }
    });
  }

  activate(r: EvacuationRoute): void { this.evacuationService.activateRoute(r.id).subscribe(() => this.load()); }
  close(r: EvacuationRoute): void { this.evacuationService.closeRoute(r.id).subscribe(() => this.load()); }
  recompute(r: EvacuationRoute): void { this.evacuationService.recomputeRoute(r.id).subscribe(() => this.load()); }

  private initPickerMap(): void {
    if (this.pickerMapInit || !this.pickerMapRef?.nativeElement) return;
    this.pickerMapInit = true;

    if (this.pickerMap) this.pickerMap.remove();
    this.pickerMap = L.map(this.pickerMapRef.nativeElement, {
      zoomControl: true, scrollWheelZoom: true, preferCanvas: true
    }).setView([36.8065, 10.1815], 10);

    L.tileLayer('https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png', {
      attribution: '© OSM © CartoDB', subdomains: 'abcd', maxZoom: 19
    }).addTo(this.pickerMap);

    this.pickerMap.getContainer().style.cursor = 'crosshair';

    this.pickerMap.on('click', (e: L.LeafletMouseEvent) => {
      const lat = parseFloat(e.latlng.lat.toFixed(6));
      const lng = parseFloat(e.latlng.lng.toFixed(6));

      if (this.pinMode === 'origin') {
        this.newRoute.fromLatitude = lat;
        this.newRoute.fromLongitude = lng;
        if (this.pickerOriginMarker) this.pickerMap.removeLayer(this.pickerOriginMarker);
        const icon = L.divIcon({
          html: `<div style="width:20px;height:20px;background:#f59e0b;border:3px solid white;border-radius:50%;box-shadow:0 0 14px rgba(245,158,11,0.9);display:flex;align-items:center;justify-content:center;font-size:10px;color:white;font-weight:bold">O</div>`,
          iconSize: [20, 20], iconAnchor: [10, 10], className: ''
        });
        this.pickerOriginMarker = L.marker([lat, lng], { icon })
          .bindPopup(`<b>Origin</b><br>${lat.toFixed(4)}, ${lng.toFixed(4)}`).addTo(this.pickerMap);
        this.pinMode = 'shelter'; // auto-switch to shelter pin next
      } else {
        this.newRoute.toLatitude = lat;
        this.newRoute.toLongitude = lng;
        if (this.pickerShelterMarker) this.pickerMap.removeLayer(this.pickerShelterMarker);
        const icon = L.divIcon({
          html: `<div style="width:20px;height:20px;background:#22c55e;border:3px solid white;border-radius:50%;box-shadow:0 0 14px rgba(34,197,94,0.9);display:flex;align-items:center;justify-content:center;font-size:10px;color:white;font-weight:bold">S</div>`,
          iconSize: [20, 20], iconAnchor: [10, 10], className: ''
        });
        this.pickerShelterMarker = L.marker([lat, lng], { icon })
          .bindPopup(`<b>Shelter</b><br>${lat.toFixed(4)}, ${lng.toFixed(4)}`).addTo(this.pickerMap);
      }
      this.cdr.detectChanges();
    });

    setTimeout(() => this.pickerMap.invalidateSize(), 100);
  }

  private initCardMaps(): void {
    this.cardMaps.forEach(m => m.remove());
    this.cardMaps.clear();

    this.filtered.forEach(r => {
      if (!r.path) return;
      const el = document.getElementById('rmap-' + r.id);
      if (!el) return;
      try {
        const map = L.map(el, {
          zoomControl: false, scrollWheelZoom: false, dragging: false,
          doubleClickZoom: false, touchZoom: false, keyboard: false,
          attributionControl: false, preferCanvas: true
        });

        L.tileLayer('https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png',
          { subdomains: 'abcd', maxZoom: 19 }).addTo(map);

        const geom = wkt.parse(r.path);
        if (geom && geom.type === 'LineString') {
          const latLngs: [number, number][] = geom.coordinates.map((c: number[]) => [c[1], c[0]]);
          const color = r.status === 'ACTIVE' ? '#22c55e' : r.status === 'BLOCKED' ? '#ef4444' : '#6c5ce7';
          const line = L.polyline(latLngs, { color, weight: 4, opacity: 0.95 }).addTo(map);

          // Origin pin (amber)
          L.circleMarker(latLngs[0], { radius: 7, color: '#f59e0b', fillColor: '#f59e0b', fillOpacity: 1, weight: 2 })
            .bindTooltip('Origin').addTo(map);
          // Shelter pin (green)
          L.circleMarker(latLngs[latLngs.length - 1], { radius: 8, color: '#22c55e', fillColor: '#22c55e', fillOpacity: 1, weight: 2 })
            .bindTooltip('Shelter').addTo(map);

          map.fitBounds(line.getBounds(), { padding: [14, 14] });
        }
        this.cardMaps.set(r.id, map);
        setTimeout(() => map.invalidateSize(), 100);
      } catch (e) { console.error('Card map error for route', r.id, e); }
    });
  }
}

  code = '';

  // Step 3

  confirmPassword = '';
  showNew = false;
  showConfirm = false;

  errorMessage = '';
  successMessage = '';

  constructor(private http: HttpClient, private router: Router) {}

  // ── Step 1 : send code ────────────────────────────────────────────────────
  sendCode(): void {
    this.errorMessage = '';
    if (!this.email.trim()) { this.errorMessage = 'Please enter your email.'; return; }

    this.isLoading = true;
    this.http.post(`${this.api}/forgot-password`, { email: this.email }, { responseType: 'text' })
      .subscribe({
        next: () => { this.isLoading = false; this.step = 'code'; },
        error: (err) => {
          this.isLoading = false;
          this.errorMessage = err?.error || 'No account found with this email.';
        }
      });
  }

  // ── Step 2 : verify code ──────────────────────────────────────────────────
  verifyCode(): void {
    this.errorMessage = '';
    if (!this.code.trim()) { this.errorMessage = 'Please enter the code.'; return; }

    this.isLoading = true;
    this.http.post(`${this.api}/verify-reset-code`, { email: this.email, code: this.code }, { responseType: 'text' })
      .subscribe({
        next: () => { this.isLoading = false; this.step = 'reset'; },
        error: (err) => {
          this.isLoading = false;
          this.errorMessage = err?.error || 'Invalid or expired code.';
        }
      });
  }

  // ── Step 3 : reset password ───────────────────────────────────────────────
  resetPassword(): void {
    this.errorMessage = '';
    if (!this.newPassword || this.newPassword.length < 6) {
      this.errorMessage = 'Password must be at least 6 characters.'; return;
    }
    if (this.newPassword !== this.confirmPassword) {
      this.errorMessage = 'Passwords do not match.'; return;
    }

    this.isLoading = true;
    this.http.post(`${this.api}/reset-password`,
      { email: this.email, code: this.code, newPassword: this.newPassword },
      { responseType: 'text' }
    ).subscribe({
      next: () => { this.isLoading = false; this.step = 'done'; },
      error: (err) => {
        this.isLoading = false;
        this.errorMessage = err?.error || 'Error resetting password.';
      }
    });
  }

  resendCode(): void {
    this.code = '';
    this.errorMessage = '';
    this.sendCode();
  }

  goToLogin(): void {
    this.router.navigateByUrl('/login');
  }
}

  styleUrls: ['./login.component.scss'],
})
export class LoginComponent {
  email: string = '';
  password: string = '';
  showPassword: boolean = false;
  isLoading: boolean = false;
  errorMessage: string = '';

  constructor(
    private authService: AuthService,
    private router: Router
  ) {}

  togglePassword(): void {
    this.showPassword = !this.showPassword;
  }

  private redirectByRole(role: string): void {
    const normalizedRole = (role || '').trim().toUpperCase();
    const adminRoles = ['PLATFORM_ADMIN', 'ORG_ADMIN', 'AGENCY_ADMIN'];

    if (adminRoles.includes(normalizedRole)) {
      this.router.navigateByUrl('/dashboard');
    } else {
      this.router.navigateByUrl('/landing');
    }
  }

  onLogin(): void {
    this.errorMessage = '';

    if (!this.email || !this.password) {
      this.errorMessage = 'Please fill in all fields.';
      return;
    }

    this.isLoading = true;

    this.authService.login({
      email: this.email,
      password: this.password
    }).subscribe({
      next: (response) => {
        this.isLoading = false;
        this.redirectByRole(response.role);
      },
      error: (error) => {
        this.isLoading = false;

        if (error?.error?.message) {
          this.errorMessage = error.error.message;
        } else if (error?.status === 401) {
          this.errorMessage = 'Invalid email or password.';
        } else if (error?.status === 0) {
          this.errorMessage = 'Cannot connect to the server.';
        } else {
          this.errorMessage = 'Login failed. Please try again.';
        }
      }
    });
  }

  onGuestLogin(): void {
    this.router.navigateByUrl('/landing');

    if (idParam === 'new') { this.isNew = true; return; }

    if (!isNaN(id)) {

export class SensorDetailComponent implements OnInit {
  sensor?: Sensor;
  isNew     = false;
  editMode  = false;
  sensorForm: FormGroup;

  constructor(
    private route: ActivatedRoute,
    private router: Router,
    private fb: FormBuilder,
    private sensorService: SensorService
  ) {
    this.sensorForm = this.fb.group({
      name:         ['', Validators.required],
      type:         ['SEISMIC', Validators.required],
      location:     ['', Validators.required],
      manufacturer: [''],
      ownerId:      [null]
    });
  }

  ngOnInit(): void {
    const idParam = this.route.snapshot.paramMap.get('id');
    if (idParam === 'new') { this.isNew = true; return; }
    const id = Number(idParam);
    if (!isNaN(id)) {
      this.sensorService.getSensorById(id).subscribe(data => {
        this.sensor = data;
        this.sensorForm.patchValue(data);
      });
    }
  }

  onLocationPicked(wkt: string): void {
    this.sensorForm.patchValue({ location: wkt });
  }

  createSensor(): void {
    if (this.sensorForm.invalid) return;
    this.sensorService.createSensor(this.sensorForm.value)
      .subscribe(() => this.router.navigate(['/sensors']));
  }

  enableEdit(): void  { this.editMode = true; }
  cancelEdit(): void  { this.editMode = false; if (this.sensor) this.sensorForm.patchValue(this.sensor); }

  updateSensor(): void {
    if (!this.sensor || this.sensorForm.invalid) return;
    this.sensorService.updateSensor(this.sensor.id, this.sensorForm.value).subscribe(updated => {
      this.sensor  = updated;
      this.editMode = false;
    });
  }

  deleteSensor(): void {
    if (!this.sensor) return;
    if (confirm('Delete this sensor permanently?')) {
      this.sensorService.deleteSensor(this.sensor.id)
        .subscribe(() => this.router.navigate(['/sensors']));
    }
  }

  ping(): void {
    if (this.sensor) {
      this.sensorService.updateLastPing(this.sensor.id).subscribe(u => this.sensor = u);
    }
  }
}

      this.sensors  = data;
      this.filtered = data;
    });
  }

  applyFilters(): void {
    this.filtered = this.sensors.filter(s => {
      const matchSearch = !this.searchTerm ||
        s.name.toLowerCase().includes(this.searchTerm.toLowerCase()) ||
        (s.manufacturer || '').toLowerCase().includes(this.searchTerm.toLowerCase());
      const matchStatus = !this.statusFilter || s.status === this.statusFilter;
      const matchType   = !this.typeFilter   || s.type   === this.typeFilter;
      return matchSearch && matchStatus && matchType;
    });
  }

      SEISMIC: 'earthquake', FLOOD_GAUGE: 'water', WEATHER: 'thunderstorm',

    if (this.map) this.map.remove();
  }

  setMode(m: QueryMode): void {
    this.mode = m;
    this.sensorResults = [];
    this.ran = false;
    this.clearResultLayers();
    if (this.map) {
      this.map.getContainer().style.cursor = m === 'nearest' ? 'crosshair' : '';
      setTimeout(() => this.map.invalidateSize(), 50);
    }
  }

  useMapBounds(): void {
    if (!this.map) return;
    const b = this.map.getBounds();
    const sw = b.getSouthWest(), ne = b.getNorthEast();
    this.polygonInput = `POLYGON((${sw.lng.toFixed(6)} ${sw.lat.toFixed(6)}, ${ne.lng.toFixed(6)} ${sw.lat.toFixed(6)}, ${ne.lng.toFixed(6)} ${ne.lat.toFixed(6)}, ${sw.lng.toFixed(6)} ${ne.lat.toFixed(6)}, ${sw.lng.toFixed(6)} ${sw.lat.toFixed(6)}))`;
  }

      next: s => {

        this.running = false;
        this.ran = true;
        this.renderPolygon(this.polygonInput);
        this.renderSensorMarkers(s, '#6c5ce7');
      },
      error: () => { this.running = false; this.ran = true; }
    });

    const point = `POINT(${this.nearLng} ${this.nearLat})`;
    this.running = true;
    this.clearResultLayers();

      next: s => {

        this.running = false;
        this.ran = true;
        this.renderOriginMarker(this.nearLat, this.nearLng);
        this.renderSensorMarkers(s, '#00b894');
        this.renderDistanceLines(s, this.nearLat, this.nearLng);
      },
      error: () => { this.running = false; this.ran = true; }
    });
  }

    if (!this.map || !s.location) return;
    try {

      if (g?.type === 'Point') this.map.setView([g.coordinates[1], g.coordinates[0]], 15, { animate: true });
    } catch {}
  }

  private initMap(): void {
    if (!this.mapRef?.nativeElement || this.map) return;

      this.nearLng = parseFloat(e.latlng.lng.toFixed(6));

        iconSize: [18, 18], iconAnchor: [9, 9], className: ''
      });

        .bindPopup(`<b>Origin</b><br>${this.nearLat.toFixed(5)}, ${this.nearLng.toFixed(5)}`)
        .addTo(this.map);
    });

    setTimeout(() => this.map.invalidateSize(), 200);
  }

  private clearResultLayers(): void {
    this.resultLayers.forEach(l => this.map?.removeLayer(l));
    this.resultLayers = [];
    if (this.originMarker) {
      this.map?.removeLayer(this.originMarker);
      this.originMarker = null;
    }
  }

  private renderSensorMarkers(sensors: SpatialSensor[], color: string): void {
    sensors.forEach(s => {
      if (!s.location) return;
      try {
        const g = wkt.parse(s.location);
        if (g?.type === 'Point') {
          const m = L.circleMarker([g.coordinates[1], g.coordinates[0]], {
            radius: 10,
            color,
            fillColor: color,
            fillOpacity: 0.85,
            weight: 2
          }).bindPopup(`<b>${s.name}</b><br><i>${s.type.replace('_', ' ')}</i><br>Status: ${s.status}`);
          m.addTo(this.map);
          this.resultLayers.push(m);
        }
      } catch {}
    });

    if (this.resultLayers.filter(l => l instanceof L.CircleMarker).length > 0) {
      const markers = this.resultLayers.filter(l => l instanceof L.CircleMarker);
      const grp = L.featureGroup(markers);
      const b = grp.getBounds();
      if (b.isValid()) this.map.fitBounds(b, { padding: [60, 60], maxZoom: 13 });
    }
  }

  private renderOriginMarker(lat: number, lng: number): void {
    if (this.originMarker) this.map.removeLayer(this.originMarker);
    const icon = L.divIcon({
      html: `<div style="width:18px;height:18px;background:#f59e0b;border:3px solid white;border-radius:50%;box-shadow:0 0 14px rgba(245,158,11,0.9)"></div>`,
      iconSize: [18, 18], iconAnchor: [9, 9], className: ''
    });
    this.originMarker = L.marker([lat, lng], { icon })
      .bindPopup(`<b>Origin</b><br>${lat.toFixed(5)}, ${lng.toFixed(5)}`)
      .addTo(this.map);
  }

  private renderDistanceLines(sensors: SpatialSensor[], lat: number, lng: number): void {
    sensors.forEach(s => {
      if (!s.location) return;
      try {
        const g = wkt.parse(s.location);
        if (g?.type === 'Point') {
          const l = L.polyline(
            [[lat, lng], [g.coordinates[1], g.coordinates[0]]],
            { color: '#00b894', weight: 1.5, dashArray: '6,5', opacity: 0.5 }
          ).addTo(this.map);
          this.resultLayers.push(l);
        }
      } catch {}
    });
  }

  private renderPolygon(polygonWkt: string): void {
    try {
      const g = wkt.parse(polygonWkt);
      if (g?.type === 'Polygon') {
        const latLngs = g.coordinates[0].map((c: number[]) => [c[1], c[0]]) as [number, number][];
        const p = L.polygon(latLngs, {
          color: '#6c5ce7',
          fillColor: '#6c5ce7',
          fillOpacity: 0.10,
          weight: 2,
          dashArray: '8,5'
        }).addTo(this.map);
        this.resultLayers.push(p);
        this.map.fitBounds(p.getBounds(), { padding: [30, 30] });
      }
    } catch {}
  }

export class MapPickerComponent implements AfterViewInit, OnDestroy {
  @ViewChild('mapEl', { static: true }) mapEl!: ElementRef;

  /** Pre-populate from an existing WKT point, e.g. "POINT(10.5 36.8)" */
  @Input() set initialWkt(wkt: string | null | undefined) {
    if (wkt) {
      this._initialWkt = wkt;
      this.pickedWkt   = wkt;
      this._tryShowPin(wkt);
    }
  }

  /** Emits a WKT string every time the user picks a point */
  @Output() locationPicked = new EventEmitter<string>();

  pickedWkt: string | null = null;
  private _initialWkt: string | null = null;
  private map!: L.Map;
  private marker?: L.Marker;

  // Default center: Tunis
  private center: [number, number] = [36.8065, 10.1815];

  ngAfterViewInit(): void {
    this.map = L.map(this.mapEl.nativeElement, {
      zoomControl: false,
      scrollWheelZoom: true,
      doubleClickZoom: false,
    }).setView(this.center, 9);

    L.control.zoom({ position: 'topright' }).addTo(this.map);

    L.tileLayer('https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png', {
      attribution: '© OSM © CartoDB',
      subdomains: 'abcd',
      maxZoom: 19,
    }).addTo(this.map);

    // Click → place / move pin
    this.map.on('click', (e: L.LeafletMouseEvent) => {
      const { lat, lng } = e.latlng;
      this._placePin(lat, lng);
      const wkt = `POINT(${lng.toFixed(6)} ${lat.toFixed(6)})`;
      this.pickedWkt = wkt;
      this.locationPicked.emit(wkt);
    });

    // If initial WKT was provided before map was ready, show pin now
    if (this._initialWkt) this._tryShowPin(this._initialWkt);

    setTimeout(() => this.map?.invalidateSize(), 100);
  }

  clearPin(): void {
    if (this.marker) { this.map.removeLayer(this.marker); this.marker = undefined; }
    this.pickedWkt = null;
    this.locationPicked.emit('');
  }

  ngOnDestroy(): void { this.map?.remove(); }

  private _placePin(lat: number, lng: number): void {
    if (this.marker) this.map.removeLayer(this.marker);
    this.marker = L.marker([lat, lng], {
      icon: L.divIcon({
        className: '',
        html: `<div style="
          width:28px;height:28px;border-radius:50% 50% 50% 0;
          background:#6C5CE7;border:3px solid #fff;
          transform:rotate(-45deg);
          box-shadow:0 2px 10px rgba(108,92,231,0.6)">
        </div>`,
        iconSize:   [28, 28],
        iconAnchor: [14, 28],
      })
    }).addTo(this.map);
  }

  private _tryShowPin(wkt: string): void {
    if (!this.map) return;
    const m = wkt.match(/POINT\(([^ ]+) ([^ )]+)\)/i);
    if (m) {
      const lng = parseFloat(m[1]);
      const lat = parseFloat(m[2]);
      this._placePin(lat, lng);
      this.map.setView([lat, lng], 13);
    }
  }

import { Component, Input, AfterViewInit, OnChanges, OnDestroy, SimpleChanges, ElementRef, ViewChild } from '@angular/core';
import { CommonModule } from '@angular/common';
import * as L from 'leaflet';
import wkt from "wkt";

@Component({
  selector: 'app-map-view',
  standalone: true,
  imports: [CommonModule],
  template: `<div #mapContainer style="height: 100%; width: 100%;"></div>`,
  styles: [`:host { display: block; height: 100%; width: 100%; }`]
})
export class MapViewComponent implements AfterViewInit, OnChanges, OnDestroy {
  @ViewChild('mapContainer', { static: true }) mapContainer!: ElementRef;
  @Input() center: [number, number] = [36.8065, 10.1815];
  @Input() zoom: number = 8;
  @Input() geometries: Array<{ type: string; wkt: string; color?: string }> = [];

  private map!: L.Map;
  private layers: L.Layer[] = [];
  private lastGeometriesHash: string = '';

  ngAfterViewInit(): void {
    this.initMap();
    this.updateGeometries();
    setTimeout(() => this.map?.invalidateSize(), 100);
  }

  ngOnChanges(changes: SimpleChanges): void {
    if (changes['geometries'] && this.map) {
      const newHash = JSON.stringify(this.geometries);
      if (newHash !== this.lastGeometriesHash) {
        this.lastGeometriesHash = newHash;
        this.updateGeometries();
      }
    }
  }

  ngOnDestroy(): void {
    if (this.map) this.map.remove();
  }

  private initMap(): void {
    this.map = L.map(this.mapContainer.nativeElement, {
      zoomControl: true,
      scrollWheelZoom: true,
      doubleClickZoom: true,
      dragging: true,
      touchZoom: true,
      boxZoom: true,
      keyboard: true
    }).setView(this.center, this.zoom);

    L.control.zoom({ position: 'topright' }).addTo(this.map);

    L.tileLayer('https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png', {
      attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OSM</a> &copy; CartoDB',
      subdomains: 'abcd',
      maxZoom: 19,
      minZoom: 1
    }).addTo(this.map);
  }

  private updateGeometries(): void {
    if (!this.map) return;
    this.clearLayers();
    this.addGeometries();
  }

  private clearLayers(): void {
    this.layers.forEach(layer => this.map?.removeLayer(layer));
    this.layers = [];
  }

  private addGeometries(): void {
    if (!this.map) return;

    this.geometries.forEach(geo => {
      try {
        const geom = wkt.parse(geo.wkt);
        let layer: L.Layer | null = null;
        if (geom.type === 'Point') {
          layer = L.marker([geom.coordinates[1], geom.coordinates[0]]);
        } else if (geom.type === 'LineString') {
          const latLngs = geom.coordinates.map((coord: number[]) => [coord[1], coord[0]]);
          layer = L.polyline(latLngs, { color: geo.color || '#3388ff', weight: 4 });
        } else if (geom.type === 'Polygon') {
          const latLngs = geom.coordinates[0].map((coord: number[]) => [coord[1], coord[0]]);
          layer = L.polygon(latLngs, { color: geo.color || '#ff3388', fillOpacity: 0.3 });
        }
        if (layer) {
          layer.addTo(this.map);
          this.layers.push(layer);
        }
      } catch (e) {
        console.error('Invalid WKT:', geo.wkt, e);
      }
    });

    // Fit bounds only on initial load, not on every update
    if (this.layers.length && this.map && !this.lastGeometriesHash) {
      const group = L.featureGroup(this.layers);
      const bounds = group.getBounds();
      if (bounds.isValid()) {
        this.map.fitBounds(bounds);
      }
    }
  }
}

import { HttpErrorResponse, HttpInterceptorFn } from '@angular/common/http';
import { Router } from '@angular/router';
import { inject } from '@angular/core';
import { catchError, throwError } from 'rxjs';

export const authInterceptor: HttpInterceptorFn = (req, next) => {
  const router = inject(Router);
  const token = localStorage.getItem('token');

  const isAuthRequest =
    req.url.includes('/auth/login') || req.url.includes('/auth/register');

  const authReq = token && !isAuthRequest
    ? req.clone({
        setHeaders: {
          Authorization: `Bearer ${token}`
        }
      })
    : req;

  return next(authReq).pipe(
    catchError((error: HttpErrorResponse) => {
      if (error.status === 401) {
        localStorage.clear();
        router.navigate(['/login']);
      }
      return throwError(() => error);
    })
  );
};

  message: string;

  password: string;

export interface RegisterRequest {
  email: string;
  password: string;
  firstName: string;
  lastName: string;
}

export class AuthService {

  constructor(private httpService: HttpService) {}

  register(data: RegisterRequest): Observable<AuthResponse> {
    return this.httpService.post<AuthResponse>('/auth/register', data).pipe(
      tap(res => this.storeAuthData(res))
    );
  }

  login(data: LoginRequest): Observable<AuthResponse> {
    return this.httpService.post<AuthResponse>('/auth/login', data).pipe(
      tap(res => this.storeAuthData(res))
    );

      tap(() => this.clearAuthData()),
      catchError((error) => {
        this.clearAuthData();
        return throwError(() => error);
      })
    );
  }

  private storeAuthData(res: AuthResponse): void {
    localStorage.setItem('token', res.token);
    localStorage.setItem('userId', res.userId.toString());
    localStorage.setItem('email', res.email);
    localStorage.setItem('role', res.role);
  }

  clearAuthData(): void {
    localStorage.removeItem('token');
    localStorage.removeItem('userId');
    localStorage.removeItem('email');
    localStorage.removeItem('role');
  }

  getToken(): string | null {
    return localStorage.getItem('token');
  }

  getUserId(): string | null {
    return localStorage.getItem('userId');
  }

  getRole(): string | null {
    return localStorage.getItem('role');
  }

  isLoggedIn(): boolean {
    return !!this.getToken();
  }
}

  private stompClient: Client | null = null;
  private messageSubject = new Subject<Message>();

  constructor(private http: HttpClient) {}

  private getAuthHeaders(): HttpHeaders {
    const token = localStorage.getItem('token');
    return new HttpHeaders({
      Authorization: `Bearer ${token}`
    });
  }

      `${this.apiUrl}/communication/conversations/me`,
      { headers: this.getAuthHeaders() }
    );

      `${this.apiUrl}/communication/messages/conversation/${conversationId}`,
      { headers: this.getAuthHeaders() }
    );
  }

      `${this.apiUrl}/communication/messages/add`,

      { headers: this.getAuthHeaders() }
    );
  }

    if (this.stompClient?.active) {
      this.stompClient.deactivate();
    }

    this.stompClient = new Client({
      webSocketFactory: () => new SockJS(`${this.apiUrl}/ws-chat`),
      reconnectDelay: 5000,
      debug: (str) => console.log('STOMP:', str)
    });

    this.stompClient.onConnect = () => {
      this.stompClient?.subscribe(
        `/topic/conversations/${conversationId}`,
        (message: IMessage) => {
          const chatMessage: Message = JSON.parse(message.body);
          this.messageSubject.next(chatMessage);
        }
      );
    };

    this.stompClient.onStompError = (frame) => {
      console.error('Broker error:', frame);
    };

    this.stompClient.activate();
  }

  sendTyping(conversationId: number): void {
    if (!this.stompClient || !this.stompClient.active) {
      return;
    }

    this.stompClient.publish({
      destination: '/app/typing',
      body: JSON.stringify({
        conversationId,
        userId: Number(localStorage.getItem('userId')),
        email: localStorage.getItem('email')
      })
    });
  }

  onTyping(conversationId: number): Observable<any> {
    return new Observable<any>((observer) => {
      let subscription: StompSubscription | null = null;

      const waitForConnection = setInterval(() => {
        if (this.stompClient && this.stompClient.active) {
          clearInterval(waitForConnection);

          subscription = this.stompClient.subscribe(
            `/topic/typing/${conversationId}`,
            (msg: IMessage) => {
              observer.next(JSON.parse(msg.body));
            }
          );
        }
      }, 200);

      return () => {
        clearInterval(waitForConnection);
        if (subscription) {
          subscription.unsubscribe();
        }
      };
    });
  }

  onMessageReceived(): Observable<Message> {
    return this.messageSubject.asObservable();
  }

  disconnect(): void {
    if (this.stompClient?.active) {
      this.stompClient.deactivate();
    }
  }
}

import { Injectable } from '@angular/core';
import { HttpClient, HttpHeaders, HttpParams } from '@angular/common/http';
import { Observable } from 'rxjs';
import { environment } from '../../../environments/environment';
import {T} from '@angular/cdk/keycodes';

@Injectable({
  providedIn: 'root'
})
export class HttpService {
  private readonly apiUrl = environment.apiUrl;

  constructor(private http: HttpClient) {}

  private buildHeaders(additional?: HttpHeaders, isFormData = false): HttpHeaders {
    const token = localStorage.getItem('token');
    let headers = new HttpHeaders();

    // Only set JSON content-type when NOT sending FormData
    if (!isFormData) {
      headers = headers.set('Content-Type', 'application/json');
      headers = headers.set('Accept', 'application/json');
    }

    if (token) {
      headers = headers.set('Authorization', `Bearer ${token}`);
    }

    if (additional) {
      additional.keys().forEach(key => {
        const val = additional.get(key);
        if (val !== null) {
          headers = headers.set(key, val);
        }
      });
    }

    return headers;
  }

  get<T>(endpoint: string, params?: { polygon: string }, customHeaders?: HttpHeaders): Observable<T> {
    const options = {
      headers: this.buildHeaders(customHeaders, false),
      params
    };
    return this.http.get<T>(`${this.apiUrl}${endpoint}`, options);
  }

  post<T>(endpoint: string, body: unknown, params?: HttpParams, customHeaders?: HttpHeaders): Observable<T> {
    const isFormData = body instanceof FormData;
    const options = {
      headers: this.buildHeaders(customHeaders, isFormData),
      params
    };
    return this.http.post<T>(`${this.apiUrl}${endpoint}`, body as any, options);
  }

  put<T>(endpoint: string, body: unknown, params?: HttpParams, customHeaders?: HttpHeaders): Observable<T> {
    const isFormData = body instanceof FormData;
    const options = {
      headers: this.buildHeaders(customHeaders, isFormData),
      params
    };
    return this.http.put<T>(`${this.apiUrl}${endpoint}`, body as any, options);
  }
  patch<T>(endpoint: string, body: unknown, params?: HttpParams, customHeaders?: HttpHeaders): Observable<T> {
    const isFormData = body instanceof FormData;
    const options = {
      headers: this.buildHeaders(customHeaders, isFormData),
      params
    };
    return this.http.patch<T>(`${this.apiUrl}${endpoint}`, body as any, options);
  }
  delete<T>(endpoint: string, params?: HttpParams, customHeaders?: HttpHeaders): Observable<T> {
    const options = {
      headers: this.buildHeaders(customHeaders, false),
      params
    };
    return this.http.delete<T>(`${this.apiUrl}${endpoint}`, options);
  }
}

import { Router } from '@angular/router';

@Component({
  selector: 'app-social-auth-success',
  standalone: true,
  template: `<p>Connexion en cours...</p>`
})
export class SocialAuthSuccessComponent implements OnInit {
  constructor(private router: Router) {}

  ngOnInit(): void {
    const params = new URLSearchParams(window.location.search);

    const token = params.get('token');
    const userId = params.get('userId');
    const email = params.get('email');
    const role = params.get('role');
    const error = params.get('error');

    if (error) {
      this.router.navigateByUrl('/login');
      return;
    }

    if (token && userId && email && role) {
      localStorage.setItem('token', token);
      localStorage.setItem('userId', userId);
      localStorage.setItem('email', email);
      localStorage.setItem('role', role);

      const normalizedRole = role.trim().toUpperCase();
      const adminRoles = ['PLATFORM_ADMIN', 'ORG_ADMIN', 'AGENCY_ADMIN'];

      if (adminRoles.includes(normalizedRole)) {
        this.router.navigateByUrl('/dashboard');
      } else {
        this.router.navigateByUrl('/landing');
      }
    } else {
      this.router.navigateByUrl('/login');
    }
  }
}
